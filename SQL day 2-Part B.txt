-- 21

from members
left join borrow on members.mem_id=borrow.mem_id

-- 22

from books
left join borrow on books.book_id=borrow.book_id

-- 23

from books
join borrow on books.book_id=borrow.book_id

order by borrow_count desc
limit 1;


-- 24

from members
join borrow on members.mem_id=borrow.mem_id
group by members.mem_id,members.mem_name
having count(borrow.book_id)>2;


-- 25
