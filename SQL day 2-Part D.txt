-- 1
with average_price as (
    select avg(price) as avg_price from books
)
select * from books
where price > (select avg_price from average_price);

-- 2
with category_average as (
    select category, avg(price) as avg_price
    from books group by category
)
select * from category_average;

-- 3
with category_average as (
    select category, avg(price) as avg_price
    from books group by category
)
select * from books
where price > (
    select avg_price from category_average
    where category_average.category = books.category
);

-- 4
with category_stock as (
    select category, sum(stock_quantity) as total_stock
    from books group by category
)
select * from category_stock;

-- 5
with category_stock as (
    select category, sum(stock_quantity) as total_stock
    from books group by category
)
select * from category_stock where total_stock > 20;

-- 6
with category_max as (
    select category, max(price) as max_price
    from books group by category
)
select * from books
where price = (
    select max_price from category_max
    where category_max.category = books.category
);

inner join members on members.mem_id = borrow.mem_id;

create procedure getmemberborrowdetails(in selected_mem_id int)
begin
    select members.mem_name, books.book_name, borrow.brw_date,
           borrow.return_date
    from members
    inner join borrow on members.mem_id = borrow.mem_id
    inner join books on books.book_id = borrow.book_id
    where members.mem_id = selected_mem_id;
end //
delimiter ;
